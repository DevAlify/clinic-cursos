# Plano de implementação: rastreamento confiável e núcleo de dados (Clinic Cursos)

> **Para agentes:** SUB-SKILL OBRIGATÓRIA: use superpowers:subagent-driven-development (recomendado) ou superpowers:executing-plans para executar este plano tarefa por tarefa. Os passos usam caixas (`- [ ]`) para acompanhamento.

**Objetivo:** entregar a demo local (núcleo de dados, cenários verificáveis e simulador), o documento de 5 páginas, o README e o registro do desafio Product Builder / FDE da Clinic Cursos.

**Arquitetura:** monólito modular em TypeScript. Um log de eventos só de acréscimo é a única fonte da verdade; pessoa, toques, oportunidades, filas e métricas são projeções refeitas a partir dele por uma função determinística. Uma porta HTTP recebe WhatsApp, formulário, QR e ERP; oito ferramentas tipadas servem igualmente a pessoas, ao simulador e aos agentes de IA.

**Stack:** Node 24.12+ ou 26 executando TypeScript direto, Hono, Zod, better-sqlite3, `node:test`, React + Vite + Tailwind + shadcn no simulador.

**Spec:** `docs/processo/2026-10-29-desenho.md` (aprovada em 29/10/2026). Quem executa lê a spec e este plano.

## Restrições globais

- Node `>=24.12` (`engines.node`). O Node executa os arquivos `.ts` direto (type stripping, estável desde 24.12.0 e 25.2.0); proibido `tsx`, `ts-node` ou etapa de compilação no servidor.
- Só sintaxe TypeScript apagável: proibidos `enum`, `namespace` com valor, parameter properties e decorators. Imports relativos com extensão `.ts`; tipos com `import type`.
- Versões exatas: hono 4.13.14, @hono/node-server 2.1.5, zod 4.6.7, better-sqlite3 13.0.3, typescript 7.0.2, @types/node 26.6.6, @types/better-sqlite3 9.6.0, react e react-dom 19.3.1, @types/react e @types/react-dom 19.3.0, vite 8.3.5, @vitejs/plugin-react 6.1.2, tailwindcss e @tailwindcss/vite 4.3.3, lucide-react 1.64.0, shadcn 4.23.0 (CLI).
- O Codex roda sem rede: toda dependência entra na Tarefa 1. Nenhuma tarefa do Codex instala pacote.
- Código, banco, rotas, comandos, ids de fila e commits em inglês. Textos de interface, alertas, motivos de fila e documentos em português.
- Dados 100% fictícios: telefones das faixas usadas em `scenarios/fixtures.ts`, e-mails `@example.com`, nomes inventados.
- O log (`events.payload`) nunca guarda telefone, e-mail, nome ou texto de conversa em claro: só hashes, referências ao cofre (`pii`), códigos e trechos curtos de evidência.
- Níveis no código: `'observed' | 'declared' | 'inferred' | 'unknown'`; na interface: observada, declarada, inferida, desconhecida.
- Regras v1: janela de atribuição de 30 dias; limite de confiança da IA 0,85; monitor com 3 últimos leads sem campanha, base de pelo menos 10 leads em 30 dias e cobertura de base de pelo menos 90%.
- Segredos de demonstração vêm de variáveis de ambiente, com padrão explícito `demo-...`. Nenhum segredo real no repositório.
- Testes, `verify` e `replay` rodam sem rede.
- Commits terminam com `Co-Authored-By: Claude Opus 5.5 <noreply@anthropic.com>` quando feitos pelo Claude.

```ts
import type { ENTRY_POINT_STATUSES, EventType, IDENTIFIER_KINDS, LEVELS, Payload, UNITS } from './events.ts';

export type Level = (typeof LEVELS)[number];
export type Unit = (typeof UNITS)[number];
export type EntryPointStatus = (typeof ENTRY_POINT_STATUSES)[number];
export type IdentifierKind = (typeof IDENTIFIER_KINDS)[number];

export type FieldName = 'entryPoint' | 'channel' | 'campaign' | 'unit' | 'product';
export interface Field {
  value: string | null;
  level: Level;
  reader: string;
  evidence?: string;
}
export type Fields = Record<FieldName, Field>;

export interface Identifier {
  kind: IdentifierKind;
  hash: string;
  piiRef: string | null;
}

export interface Rules {
  attributionWindowDays: number;
  aiConfidenceThreshold: number;
}

export interface EventRow<T extends EventType = EventType> {
  seq: number;
  id: string;
  source: string;
  externalId: string;
  type: T;
  schemaVersion: number;
  occurredAt: string;
  receivedAt: string;
  actor: string;
  payload: Payload<T>;
}
export type AnyEvent = { [T in EventType]: EventRow<T> }[EventType];

export interface EventInput<T extends EventType = EventType> {
  source: string;
  externalId: string;
  type: T;
  occurredAt: string;
  actor: string;
  payload: Payload<T>;
}
export type AnyEventInput = { [T in EventType]: EventInput<T> }[EventType];
```

```ts
import { test } from 'node:test';
import assert from 'node:assert/strict';
import { openDb, resetProjections } from '../src/core/db.ts';
import { appendEvent, listEvents } from '../src/core/log.ts';

const scan = (externalId: string) => ({
  source: 'qr', externalId, type: 'qr_scanned' as const, occurredAt: '2026-10-29T14:00:00-03:00',
  actor: 'system', payload: { code: 'K7P2' },
});

test('appendEvent stores the event in UTC with a sequence number', () => {
  const db = openDb();
  const r = appendEvent(db, scan('s1'));
  assert.equal(r.duplicate, false);
  assert.equal(r.event?.seq, 1);
  assert.equal(listEvents(db)[0].occurredAt, '2026-10-29T17:00:00.000Z');
});

test('the same source key is stored once and the repeat is counted', () => {
  const db = openDb();
  appendEvent(db, scan('s1'));
  assert.equal(appendEvent(db, scan('s1')).duplicate, true);
  assert.equal(listEvents(db).length, 1);
  assert.deepEqual(db.prepare('SELECT count FROM inbound_duplicates WHERE external_id = ?').get('s1'), { count: 1 });
});

test('the log rejects update and delete', () => {
  const db = openDb();
  appendEvent(db, scan('s1'));
  assert.throws(() => db.prepare("UPDATE events SET actor = 'x'").run(), /append-only/);
  assert.throws(() => db.prepare('DELETE FROM events').run(), /append-only/);
});

test('invalid payloads are refused before storage', () => {
  const db = openDb();
  assert.throws(() => appendEvent(db, { ...scan('s1'), payload: {} as never }));
  assert.equal(listEvents(db).length, 0);
});

test('resetProjections keeps the log and the vault', () => {
  const db = openDb();
  appendEvent(db, scan('s1'));
  db.prepare("INSERT INTO pii (ref, kind, value) VALUES ('pii_x', 'name', 'Ana')").run();
  db.prepare("INSERT INTO scans (code, count) VALUES ('K7P2', 1)").run();
  resetProjections(db);
```

```ts
import { payloadSchemas, type EventType, type Payload } from './events.ts';
import type { AnyEvent, EventInput, EventRow } from './types.ts';

export type AppendResult<T extends EventType> =
  | { event: EventRow<T>; duplicate: false }
  | { event: null; duplicate: true };

function toUtcIso(value: string): string {
  const time = Date.parse(value);
  if (Number.isNaN(time)) throw new Error(`data inválida: ${value}`);
  return new Date(time).toISOString();
}

export function appendEvent<T extends EventType>(
  db: Db, input: EventInput<T>, receivedAt = new Date().toISOString(),
): AppendResult<T> {
  const payload = payloadSchemas[input.type].parse(input.payload) as Payload<T>;
  const occurredAt = toUtcIso(input.occurredAt);
  const id = randomUUID();
  const inserted = db
    .prepare(`INSERT INTO events (id, source, external_id, type, schema_version, occurred_at, received_at, actor, payload)
      VALUES (?, ?, ?, ?, 1, ?, ?, ?, ?) ON CONFLICT (source, external_id) DO NOTHING`)
    .run(id, input.source, input.externalId, input.type, occurredAt, receivedAt, input.actor, JSON.stringify(payload));
  if (inserted.changes === 0) {
    db.prepare(`INSERT INTO inbound_duplicates (source, external_id, count, last_seen_at) VALUES (?, ?, 1, ?)
      ON CONFLICT (source, external_id) DO UPDATE SET count = count + 1, last_seen_at = excluded.last_seen_at`)
      .run(input.source, input.externalId, receivedAt);
    return { event: null, duplicate: true };
  }
  const event: EventRow<T> = {
    seq: Number(inserted.lastInsertRowid), id, source: input.source, externalId: input.externalId,
    type: input.type, schemaVersion: 1, occurredAt, receivedAt, actor: input.actor, payload,
  };
  return { event, duplicate: false };
}
```

```ts
import { createHmac } from 'node:crypto';
import { SECRETS } from '../config.ts';
import type { Db } from './db.ts';
import type { Identifier, IdentifierKind } from './types.ts';

// Brazilian mobiles: an 8-digit local number starting with 6-9 is the old format and gets the leading 9.
export function normalizePhone(raw: string, hasCountryCode: boolean): string | null {
  let digits = raw.replace(/\D/g, '');
  if (!hasCountryCode && (digits.length === 10 || digits.length === 11)) digits = `55${digits}`;
  if (!digits.startsWith('55')) return digits.length >= 8 && digits.length <= 15 ? `+${digits}` : null;
  const rest = digits.slice(2);
  if (rest.length !== 10 && rest.length !== 11) return null;
  const ddd = rest.slice(0, 2);
  let local = rest.slice(2);
  if (local.length === 8 && /^[6-9]/.test(local)) local = `9${local}`;
  return `+55${ddd}${local}`;
}

export function normalizeEmail(raw: string): string | null {
  const email = raw.trim().toLowerCase();
  return /^[^@\s]+@[^@\s]+\.[^@\s]+$/.test(email) ? email : null;
}

export function hashIdentifier(kind: string, value: string): string {
  return createHmac('sha256', SECRETS.piiHashKey).update(`${kind}:${value}`).digest('hex');
}

export function storePii(db: Db, kind: string, value: string): string {
  const ref = `pii_${hashIdentifier(kind, value).slice(0, 16)}`;
  db.prepare('INSERT OR IGNORE INTO pii (ref, kind, value) VALUES (?, ?, ?)').run(ref, kind, value);
  return ref;
}

export function readPii(db: Db, ref: string | null): string | null {
  if (!ref) return null;
  const row = db.prepare('SELECT value FROM pii WHERE ref = ?').get(ref) as { value: string } | undefined;
  return row?.value ?? null;
}

export function toIdentifier(db: Db, kind: IdentifierKind, raw: string, hasCountryCode = false): Identifier | null {
  const value = kind === 'phone' ? normalizePhone(raw, hasCountryCode) : kind === 'email' ? normalizeEmail(raw) : raw.trim();
  if (!value) return null;
  return { kind, hash: hashIdentifier(kind, value), piiRef: storePii(db, kind, value) };
}
```

```ts
import type { Db } from './db.ts';
import type { Identifier, IdentifierKind } from './types.ts';

export const IDENTITY_PRIORITY: IdentifierKind[] = ['bsuid', 'phone', 'email', 'erp_student', 'erp_patient'];

export interface Resolution {
  personId: string;
  created: boolean;
  conflictWith: string[];
}

const rank = (kind: string) => IDENTITY_PRIORITY.indexOf(kind as IdentifierKind);

function ownerOf(db: Db, kind: string, hash: string): string | null {
  const row = db.prepare('SELECT person_id FROM identifiers WHERE kind = ? AND hash = ?').get(kind, hash) as
    | { person_id: string }
    | undefined;
  return row?.person_id ?? null;
}

export function findPerson(db: Db, identifiers: { kind: string; hash: string }[]): string | null {
  for (const i of [...identifiers].sort((a, b) => rank(a.kind) - rank(b.kind))) {
    const owner = ownerOf(db, i.kind, i.hash);
    if (owner) return owner;
  }
  return null;
}

// Exact matches only. On conflict, the strongest identifier wins and the other person is reported for review.
export function resolvePerson(db: Db, identifiers: Identifier[], seq: number, displayRef: string | null): Resolution {
  const found = [...identifiers]
    .sort((a, b) => rank(a.kind) - rank(b.kind))
    .map((identifier) => ({ identifier, owner: ownerOf(db, identifier.kind, identifier.hash) }));
  const owners = [...new Set(found.flatMap((f) => (f.owner ? [f.owner] : [])))];
  const created = owners.length === 0;
  const personId = created ? `P${seq}` : owners[0];
  if (created) {
    db.prepare('INSERT INTO people (id, created_seq, display_ref) VALUES (?, ?, ?)').run(personId, seq, displayRef);
  } else if (displayRef) {
    db.prepare('UPDATE people SET display_ref = COALESCE(display_ref, ?) WHERE id = ?').run(displayRef, personId);
  }
  for (const f of found) {
    if (!f.owner) {
      db.prepare('INSERT INTO identifiers (kind, hash, person_id, pii_ref) VALUES (?, ?, ?, ?)')
        .run(f.identifier.kind, f.identifier.hash, personId, f.identifier.piiRef);
    }
  }
  return { personId, created, conflictWith: owners.filter((o) => o !== personId) };
}
```

```ts
import { randomInt } from 'node:crypto';

// No 0, O, 1, I or L: nobody confuses them when typing or reading aloud.
export const CODE_ALPHABET = 'ABCDEFGHJKMNPQRSTUVWXYZ23456789';

export function newCode(exists: (code: string) => boolean): string {
  for (;;) {
    let code = '';
    for (let i = 0; i < 4; i++) code += CODE_ALPHABET[randomInt(CODE_ALPHABET.length)];
    if (!exists(code)) return code;
  }
}
```

```ts
import type { Level } from './types.ts';

export interface TouchLite { seq: number; occurredAt: string; channelLevel: Level; isTest: boolean }
export interface Attribution { firstSeq: number | null; lastSeq: number | null; rule: string }

const DAY_MS = 86_400_000;

export const attributionRule = (windowDays: number): string => `known-channel-touch-${windowDays}d@v1`;

// Linking a sale to a touch is a conclusion: named rule, version and window, by event time.
export function computeAttribution(touches: TouchLite[], openedAt: string, windowDays: number): Attribution {
  const end = Date.parse(openedAt);
  const start = end - windowDays * DAY_MS;
  const candidates = touches
    .filter((t) => !t.isTest && t.channelLevel !== 'unknown')
    .filter((t) => {
      const at = Date.parse(t.occurredAt);
      return at >= start && at <= end;
    })
    .sort((a, b) => Date.parse(a.occurredAt) - Date.parse(b.occurredAt) || a.seq - b.seq);
  return { firstSeq: candidates[0]?.seq ?? null, lastSeq: candidates.at(-1)?.seq ?? null, rule: attributionRule(windowDays) };
}
```

```ts
import { MONITOR } from '../config.ts';
import type { Level } from './types.ts';

export interface PointTouch { seq: number; occurredAt: string; campaignLevel: Level; isTest: boolean }

const DAY_MS = 86_400_000;
const byTime = (a: PointTouch, b: PointTouch) => Date.parse(a.occurredAt) - Date.parse(b.occurredAt) || a.seq - b.seq;

// Returns the affected leads when the last N arrived without campaign after a healthy baseline.
export function findCoverageDrop(points: PointTouch[], cfg = MONITOR): PointTouch[] | null {
  const real = points.filter((p) => !p.isTest).sort(byTime);
  if (real.length <= cfg.lastLeads) return null;
  const last = real.slice(-cfg.lastLeads);
  if (last.some((p) => p.campaignLevel !== 'unknown')) return null;
  const from = Date.parse(last[0].occurredAt) - cfg.baselineDays * DAY_MS;
  const base = real.slice(0, -cfg.lastLeads).filter((p) => Date.parse(p.occurredAt) >= from);
  if (base.length < cfg.minBaselineLeads) return null;
  const coverage = base.filter((p) => p.campaignLevel !== 'unknown').length / base.length;
  return coverage >= cfg.minBaselineCoverage ? last : null;
}

export function alertText(label: string, firstAffectedAt: string, affected: number, owner: string): { message: string; action: string } {
  const time = new Date(firstAffectedAt)
    .toLocaleTimeString('pt-BR', { timeZone: 'America/Sao_Paulo', hour: '2-digit', minute: '2-digit' })
    .replace(':', 'h');
  const subject = label.charAt(0).toUpperCase() + label.slice(1);
  return {
    message: `${subject} parou de enviar a campanha desde as ${time}. ${affected} leads afetados.`,
    action: `Conferir a página publicada. Responsável: ${owner}.`,
  };
}
```

```ts
// The person's open WhatsApp conversation: latest WhatsApp touch in the last 24 hours.
export function conversationTouch(db: Db, personId: string, at: string, hours = CONVERSATION_WINDOW_HOURS): TouchRow | null {
  const end = Date.parse(at);
  const start = end - hours * 3_600_000;
  const touches = touchesOfPerson(db, personId).filter((t) => t.source === 'whatsapp');
  return touches.reverse().find((t) => Date.parse(t.occurredAt) <= end && Date.parse(t.occurredAt) >= start) ?? null;
}
```

```ts
// Deterministic: no clock, no network, no AI. Same log + same rules = same projections.
export function project(db: Db, e: AnyEvent, rules: Rules): void {
  switch (e.type) {
    case 'entry_point_registered': {
      const p = e.payload;
      db.prepare('INSERT OR REPLACE INTO entry_points (code, unit, kind, label, owner, status, product, business_number) VALUES (?, ?, ?, ?, ?, ?, ?, ?)')
        .run(p.code, p.unit, p.kind, p.label, p.owner, p.status, p.product, p.businessNumber);
      return;
    }
    case 'entry_point_status_changed':
      db.prepare('UPDATE entry_points SET status = ? WHERE code = ?').run(e.payload.status, e.payload.code);
      return;
    case 'link_generated': {
      const p = e.payload;
      db.prepare('INSERT OR REPLACE INTO links (code, entry_point, channel, campaign, unit, product, destination, link_type) VALUES (?, ?, ?, ?, ?, ?, ?, ?)')
        .run(p.code, p.entryPoint, p.channel, p.campaign, p.unit, p.product, p.destination, p.linkType);
      return;
    }
    case 'qr_scanned':
      db.prepare('INSERT INTO scans (code, count) VALUES (?, 1) ON CONFLICT (code) DO UPDATE SET count = count + 1').run(e.payload.code);
      return;
    case 'form_submitted': {
      const p = e.payload;
      const personId = person(db, e, p.identifiers, p.displayRef);
      const ep = getEntryPoint(db, p.entryPoint);
      insertTouch(db, e, personId, 'form', originFromForm(p, ep), null, p.test);
      if (!p.test && ep?.product && isCatalogProduct(ep.product)) ensureOpportunity(db, e, personId, ep.product, ep.unit, rules);
      recomputeAttributions(db, personId, rules);
      return;
    }
    case 'whatsapp_message_received': {
      const p = e.payload;
      const personId = person(db, e, p.identifiers, p.displayRef);
      const link = p.code ? getLink(db, p.code) : null;
      if (link || p.referral || !conversationTouch(db, personId, e.occurredAt)) {
        insertTouch(db, e, personId, 'whatsapp', originFromWhatsApp(p, getEntryPointByNumber(db, p.businessNumber), link), p.code, false);
        if (link && isCatalogProduct(link.product)) ensureOpportunity(db, e, personId, link.product, link.unit, rules);
      }
      recomputeAttributions(db, personId, rules);
      return;
    }
    case 'erp_person_imported': {
      const p = e.payload;
      const personId = person(db, e, p.identifiers, p.displayRef);
      ensureRole(db, personId, p.role, p.unit, e.seq);
      insertTouch(db, e, personId, 'erp', unknownFields(`import:${p.batch}`), null, false);
      return;
    }
    case 'erp_sale_confirmed': {
      const p = e.payload;
      const personId = person(db, e, p.identifiers, p.displayRef);
      ensureRole(db, personId, 'student', p.unit, e.seq);
      const opp = p.opportunityId ? getOpportunity(db, p.opportunityId) : null;
      if (opp) {
        db.prepare("UPDATE opportunities SET status = 'won', erp_sale_id = ? WHERE id = ?").run(p.saleId, opp.id);
      } else {
        const suggestion = findOpenOpportunity(db, personId, p.product, p.unit)?.id ?? null;
        enqueue(db, e, 'reconciliation', 'reconciliation', personId, 'Venda do ERP sem oportunidade ligada', { saleId: p.saleId, product: p.product, unit: p.unit, suggestion });
      }
      return;
    }
    case 'conversation_routed': {
      const p = e.payload;
      enqueue(db, e, 'routing', p.queue, p.personId, p.reason, { intent: p.intent, confidence: p.confidence, model: p.model, summary: p.summary, messageSeq: p.messageSeq });
      return;
    }
    case 'origin_declared': {
      const p = e.payload;
      const touch = getTouch(db, p.touchSeq);
      if (!touch) throw new Error(`toque ${p.touchSeq} não existe`);
      if (accepted(e.actor, p.confidence, rules)) {
        db.prepare('UPDATE touches SET fields = ? WHERE seq = ?').run(JSON.stringify(withDeclaredChannel(touch.fields, p.value, e.actor, p.quote)), touch.seq);
        recomputeAttributions(db, touch.personId, rules);
      } else {
        enqueue(db, e, 'origin_review', 'triage', p.personId, `Origem declarada com confiança baixa (${p.confidence})`, { touchSeq: p.touchSeq, value: p.value, quote: p.quote });
      }
      return;
    }
    case 'interest_declared': {
      const p = e.payload;
      if (accepted(e.actor, p.confidence, rules) && isCatalogProduct(p.product)) {
        ensureOpportunity(db, e, p.personId, p.product, p.unit, rules);
        recomputeAttributions(db, p.personId, rules);
      } else {
        enqueue(db, e, 'interest_review', 'triage', p.personId, `Interesse a confirmar: ${p.product} · ${p.unit}`, { product: p.product, unit: p.unit, confidence: p.confidence });
      }
      return;
    }
    case 'merge_proposed':
      enqueue(db, e, 'merge_review', 'identity-review', e.payload.personIds[0], e.payload.reason, { personIds: e.payload.personIds });
      return;
    case 'alert_raised': {
      const p = e.payload;
      db.prepare('INSERT INTO alerts (entry_point, kind, message, action, owner, created_seq) VALUES (?, ?, ?, ?, ?, ?)')
        .run(p.entryPoint, p.kind, p.message, p.action, p.owner, e.seq);
      return;
    }
  }
}

const accepted = (actor: string, confidence: number | null, rules: Rules) =>
  !actor.startsWith('ai:') || (confidence ?? 0) >= rules.aiConfidenceThreshold;

function person(db: Db, e: AnyEvent, identifiers: Identifier[], displayRef: string | null): string {
  const r = resolvePerson(db, identifiers, e.seq, displayRef);
  for (const other of r.conflictWith) {
    enqueue(db, e, 'merge_review', 'identity-review', r.personId, 'Identificadores apontam para fichas diferentes', { personIds: [r.personId, other] });
  }
  return r.personId;
}

function insertTouch(db: Db, e: AnyEvent, personId: string, source: string, fields: Fields, code: string | null, isTest: boolean): void {
  db.prepare('INSERT INTO touches (seq, person_id, occurred_at, source, fields, code, is_test) VALUES (?, ?, ?, ?, ?, ?, ?)')
    .run(e.seq, personId, e.occurredAt, source, JSON.stringify(fields), code, isTest ? 1 : 0);
}

function ensureRole(db: Db, personId: string, role: string, unit: string, seq: number): void {
  db.prepare('INSERT OR IGNORE INTO roles (person_id, role, unit, source_seq) VALUES (?, ?, ?, ?)').run(personId, role, unit, seq);
}

function ensureOpportunity(db: Db, e: AnyEvent, personId: string, product: string, unit: string, rules: Rules): string {
  const open = findOpenOpportunity(db, personId, product, unit);
  if (open) return open.id;
  const id = `O${e.seq}`;
  db.prepare("INSERT INTO opportunities (id, person_id, product, unit, status, opened_seq, opened_at, attribution_rule) VALUES (?, ?, ?, ?, 'open', ?, ?, ?)")
    .run(id, personId, product, unit, e.seq, e.occurredAt, attributionRule(rules.attributionWindowDays));
  ensureRole(db, personId, 'lead', unit, e.seq);
  const pending = db.prepare("SELECT id, details FROM queue_items WHERE kind = 'reconciliation' AND status = 'open' AND person_id = ?").all(personId) as { id: number; details: string }[];
  for (const item of pending) {
    const d = JSON.parse(item.details) as { product: string; unit: string; suggestion: string | null };
    if (d.product === product && d.unit === unit && !d.suggestion) {
      db.prepare('UPDATE queue_items SET details = ? WHERE id = ?').run(JSON.stringify({ ...d, suggestion: id }), item.id);
    }
  }
  return id;
}

function recomputeAttributions(db: Db, personId: string, rules: Rules): void {
  const touches = touchesOfPerson(db, personId).map((t) => ({ seq: t.seq, occurredAt: t.occurredAt, channelLevel: t.fields.channel.level, isTest: t.isTest }));
  const opps = db.prepare('SELECT id, opened_at FROM opportunities WHERE person_id = ?').all(personId) as { id: string; opened_at: string }[];
  for (const o of opps) {
    const a = computeAttribution(touches, o.opened_at, rules.attributionWindowDays);
    db.prepare('UPDATE opportunities SET first_touch_seq = ?, last_touch_seq = ?, attribution_rule = ? WHERE id = ?').run(a.firstSeq, a.lastSeq, a.rule, o.id);
  }
}

function enqueue(db: Db, e: AnyEvent, kind: string, queue: string, personId: string | null, reason: string, details: object): void {
  db.prepare('INSERT INTO queue_items (kind, queue, person_id, reason, details, created_seq) VALUES (?, ?, ?, ?, ?, ?)')
    .run(kind, queue, personId, reason, JSON.stringify(details), e.seq);
}
```

```ts
import type { Db } from './db.ts';
import { appendEvent } from './log.ts';
import { project } from './project.ts';
import type { AnyEvent, AnyEventInput, Rules } from './types.ts';

export interface ApplyResult { event: AnyEvent | null; duplicate: boolean; error: string | null }

// Store first, then project. A projection error never loses the event: it goes to quarantine.
export function appendAndProject(db: Db, input: AnyEventInput, rules: Rules): ApplyResult {
  const appended = appendEvent(db, input);
  if (appended.duplicate) return { event: null, duplicate: true, error: null };
  const event = appended.event as AnyEvent;
  try {
    db.transaction(() => project(db, event, rules))();
    return { event, duplicate: false, error: null };
  } catch (err) {
    const error = err instanceof Error ? err.message : String(err);
    db.prepare('INSERT OR REPLACE INTO quarantine (seq, error) VALUES (?, ?)').run(event.seq, error);
    return { event, duplicate: false, error };
  }
}
```

```ts
export function rebuild(db: Db, rules: Rules): void {
  resetProjections(db);
  for (const e of listEvents(db)) {
    try {
      db.transaction(() => project(db, e, rules))();
    } catch (err) {
      db.prepare('INSERT OR REPLACE INTO quarantine (seq, error) VALUES (?, ?)').run(e.seq, err instanceof Error ? err.message : String(err));
    }
  }
}

function summary(db: Db, seq: number | null): TouchSummary | null {

}

// Rebuilds an in-memory copy with the new rules and lists what changes. The live database is untouched.
export function previewRules(db: Db, next: Rules): AttributionChange[] {
  const copy = new Database(db.serialize());
  try {
    rebuild(copy, next);
    return allOpportunities(db).flatMap((before) => {
      const after = getOpportunity(copy, before.id);
      if (after?.firstTouchSeq === before.firstTouchSeq) return [];
      return [{
        opportunityId: before.id, personName: personName(db, before.personId), product: before.product, unit: before.unit,
        before: summary(db, before.firstTouchSeq), after: after ? summary(copy, after.firstTouchSeq) : null,
      }];
    });
  } finally {
    copy.close();
  }
}
```

```ts
export function getCoverage(db: Db): Coverage {
  const byUnit: Record<string, LevelMix> = {};
  const byChannel: Record<string, number> = {};
  const byCampaign: Record<string, number> = {};
  const released = new Set(allEntryPoints(db).filter((ep) => ep.status === 'released').map((ep) => ep.code));
  for (const o of allOpportunities(db)) {
    const t = o.firstTouchSeq === null ? null : getTouch(db, o.firstTouchSeq);
    const level: Level = t?.fields.channel.level ?? 'unknown';
    const mix = (byUnit[o.unit] ??= { total: 0, observed: 0, declared: 0, inferred: 0, unknown: 0 });
    mix.total += 1;
    mix[level] += 1;
    const channel = t?.fields.channel.value ?? 'desconhecido';
    byChannel[channel] = (byChannel[channel] ?? 0) + 1;
    // Campaign numbers count only leads from released entry points.
    const campaign = t?.fields.campaign.value;
    const point = t?.fields.entryPoint.value;
    if (campaign && point && released.has(point)) byCampaign[campaign] = (byCampaign[campaign] ?? 0) + 1;
  }
  const byEntryPoint = allEntryPoints(db).map((ep) => {
    const touches = touchesOfEntryPoint(db, ep.code).filter((t) => !t.isTest);
    return { code: ep.code, label: ep.label, status: ep.status, total: touches.length, campaignObserved: touches.filter((t) => t.fields.campaign.level === 'observed').length };
  });
  return { byUnit, byChannel, byCampaign, byEntryPoint };
}
```

```ts
export type Intent = 'course_purchase' | 'patient_booking' | 'student_question' | 'complaint' | 'clinical_question' | 'other';

export interface Classification {
  intent: Intent;
  confidence: number;
  declaredChannel: { value: string; quote: string; confidence: number } | null;
  product: { value: string; confidence: number } | null;
  model: string;
}

export const MODEL = 'simulated-v1';

export const INTENT_LABEL: Record<Intent, string> = {
  course_purchase: 'Quer comprar um curso', patient_booking: 'Quer ser paciente modelo', student_question: 'Dúvida de aluno',
  complaint: 'Reclamação', clinical_question: 'Dúvida clínica', other: 'Outro assunto',
};

// Order matters: sensitive subjects win over sales words in the same message.
const INTENT_RULES: [Intent, RegExp][] = [
  ['clinical_question', /\b(dor|inchad\w*|incha[çc]o|hematoma|rea[çc][ãa]o|alergia|febre)\b/i],
  ['complaint', /\b(reclama\w*|insatisfeit\w*|p[ée]ssim\w*|absurd\w*|procon)\b/i],
  ['patient_booking', /\b(paciente|modelo|agendar)\b/i],
  ['student_question', /\b(aluno|aluna|certificado|apostila)\b/i],
  ['course_purchase', /\b(cursos?|turma|matr[ií]cula|inscrever|inscri[çc][ãa]o|valor|pre[çc]o|quanto custa)\b/i],
];
// The quote keeps only the phrase that proves the channel ("vi no insta"), never the rest of the message.
const said = (words: string) => new RegExp(String.raw`\b(?:(?:vi|conheci|achei)\s+(?:no|na|pelo|pela)\s+)?(?:${words})\b`, 'i');
const CHANNEL_RULES: [string, RegExp][] = [
  ['instagram', said('insta|instagram')], ['google', said('google')], ['referral', said('indica[çc][ãa]o|indicou')],
  ['tiktok', said('tiktok')], ['youtube', said('youtube')],
];
const PRODUCT_RULES: [string, RegExp][] = [['hof', /\bharmoniza[çc][ãa]o\b/i]];

// Stand-in for a real model: same input and output contract, so swapping the provider changes only this file.
export function classifyMessage(text: string): Classification {
  const intent = INTENT_RULES.find(([, re]) => re.test(text));
  const channel = CHANNEL_RULES.find(([, re]) => re.test(text));
  const product = PRODUCT_RULES.find(([, re]) => re.test(text));
  return {
    intent: intent?.[0] ?? 'other',
    confidence: intent ? 0.91 : 0.5,
    declaredChannel: channel ? { value: channel[0], quote: text.match(channel[1])![0], confidence: 0.93 } : null,
    product: product ? { value: product[0], confidence: 0.9 } : null,
    model: MODEL,
  };
}

export function routeFor(intent: Intent, unit: string | null, confidence: number, threshold: number): { queue: string; why: string } {
  if (confidence < threshold || intent === 'other') return { queue: 'triage', why: 'a IA não tem certeza do assunto' };
  if (intent === 'complaint') return { queue: 'human-priority', why: 'reclamação' };
  if (intent === 'clinical_question') return { queue: 'clinical-team', why: 'dúvida clínica, que a IA não responde' };
  if (intent === 'student_question') return { queue: 'student-support', why: 'dúvida de aluno' };
  if (!unit) return { queue: 'triage', why: 'unidade desconhecida' };
  return intent === 'course_purchase' ? { queue: `sales-${unit}`, why: 'interesse em curso' } : { queue: `patients-${unit}`, why: 'paciente modelo' };
}
```

```ts
  try {
    return { ok: true, result: tool.run({ db, actor, rules, now }, parsed.data as never) };
  } catch (err) {
    return { ok: false, error: err instanceof Error ? err.message : String(err) };
  }
```

```ts
import { isValidCampaign } from './naming.ts';
import { getEntryPoint, getTouch, touchesOfEntryPoint } from './queries.ts';
import type { AnyEventInput, EventRow, Rules } from './types.ts';

// Live entry point: store, project, then react. Reactions write new events; replay only projects.
export function ingest(db: Db, input: AnyEventInput, rules: Rules, transient: { text?: string } = {}): ApplyResult {
  const r = appendAndProject(db, input, rules);
  if (!r.event || r.error) return r;
  if (r.event.type === 'whatsapp_message_received') aiReaction(db, r.event as EventRow<'whatsapp_message_received'>, transient.text ?? '', rules);
  if (r.event.type === 'form_submitted' || r.event.type === 'whatsapp_message_received') monitorReaction(db, r.event.seq, r.event.occurredAt, rules);
  return r;
}

function raise(db: Db, seq: number, occurredAt: string, payload: Payload<'alert_raised'>, rules: Rules): void {
  appendAndProject(db, { source: 'monitor', externalId: `alert:${payload.kind}:${payload.entryPoint}:${seq}`, type: 'alert_raised', occurredAt, actor: 'monitor', payload }, rules);
}

export function monitorReaction(db: Db, seq: number, occurredAt: string, rules: Rules): void {
  const touch = getTouch(db, seq);
  const code = touch?.fields.entryPoint.value;
  if (!touch || touch.isTest || !code) return;
  const ep = getEntryPoint(db, code);
  if (!ep) return;
  const campaign = touch.fields.campaign;
  if (campaign.level === 'observed' && campaign.value && !isValidCampaign(campaign.value)) {
    raise(db, seq, occurredAt, { entryPoint: code, kind: 'naming_nonconforming', message: `Campanha fora do padrão: ${campaign.value}`, action: 'Gerar o link pelo gerador, com valores do catálogo.', owner: ep.owner }, rules);
  }
  if (ep.status !== 'released') return;
  const points = touchesOfEntryPoint(db, code).map((t) => ({ seq: t.seq, occurredAt: t.occurredAt, campaignLevel: t.fields.campaign.level, isTest: t.isTest }));
  const affected = findCoverageDrop(points);
  if (!affected) return;
  const text = alertText(ep.label, affected[0].occurredAt, affected.length, ep.owner);
  raise(db, seq, occurredAt, { entryPoint: code, kind: 'campaign_coverage_drop', message: text.message, action: text.action, owner: ep.owner }, rules);
  callTool(db, 'set_entry_point_status', { code, status: 'auditing', reason: text.message }, 'monitor', rules, occurredAt);
}
```

```ts
import { createHmac, timingSafeEqual } from 'node:crypto';

export const signMeta = (rawBody: string, secret: string): string =>
  `sha256=${createHmac('sha256', secret).update(rawBody, 'utf8').digest('hex')}`;

export function safeEqual(a: string, b: string): boolean {
  const x = Buffer.from(a);
  const y = Buffer.from(b);
  return x.length === y.length && timingSafeEqual(x, y);
}

// Checked against the raw body, before any JSON parsing.
export const verifyMetaSignature = (rawBody: string, header: string | undefined, secret: string): boolean =>
  header !== undefined && safeEqual(header, signMeta(rawBody, secret));
```

```ts
      out.push({
```

```ts
    ran.add(step.id);
    return c.json({ ok: res.status < 400, status: res.status });
  });

  app.post('/api/reset', (c) => {
    db.close();
    db = opts.openFresh();
    rules = initialRules();
    seed(db, rules);
    ran.clear();
    return c.json({ ok: true });
  });

    } catch (err) {
      return c.json({ error: err instanceof Error ? err.message : String(err) }, 400);
    }
  });

  return { app, getDb: () => db };
```

```ts
import { DEFAULT_RULES } from '../src/config.ts';
import { openDb, type Db } from '../src/core/db.ts';
import { getState } from '../src/core/state.ts';
import type { Rules } from '../src/core/types.ts';
import { createApp } from '../src/server.ts';
import { checks, type Check } from './checks.ts';
import { steps, type Step } from './steps.ts';

export interface ScenarioRun {
  db: Db; rules: Rules;
  outcomes: { step: Step; status: number }[];
  results: { check: Check; ok: boolean; actual: string }[];
}

// Same path as the simulator: every step goes through the HTTP door of the real app.
export async function runScenario(rules: Rules = DEFAULT_RULES): Promise<ScenarioRun> {
  const { app, getDb } = createApp({ openFresh: () => openDb(), rules });
  const run: ScenarioRun = { db: getDb(), rules, outcomes: [], results: [] };
  const verify = (stepId: string) => {
    const db = getDb();
    const state = getState(db, rules);
    for (const check of checks.filter((c) => c.afterStep === stepId)) {
      try {
        run.results.push({ check, ...check.run({ db, state, rules }) });
      } catch (err) {
        run.results.push({ check, ok: false, actual: `erro: ${err instanceof Error ? err.message : String(err)}` });
      }
    }
  };
  for (const step of steps) {
    const req = step.build({ db: getDb() });
    const res = await app.request(req.path, { method: req.method, headers: req.headers, body: req.body });
    run.outcomes.push({ step, status: res.status });
    verify(step.id);
  }
  verify('end');
  run.db = getDb();
  return run;
}
```
